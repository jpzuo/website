### this 指向的四种情况

1. 默认指向

```js
function foo() {
  console.log(this.a);
}
var a = 1;
foo(); // 1
```2. 隐式绑定

```js
function foo() {
  console.log(this.a);
}
var obj = { a: 2, foo }; // 隐式绑定
obj.foo(); // 2
```3. 显式绑定

```js
function foo() {
  console.log(this.a);
}
var obj = { a: 2 };
var bar = function () {
  foo.call(obj);
}; // 显式绑定
bar(); // 2
```4. new 绑定

```js
function foo(a) {
  this.a = a;
}
var bar = new foo(2);
console.log(bar.a); // 2
```

### this 指向优先级

new 绑定 > 显式绑定 > 隐式绑定 > 默认绑定

### 箭头函数

箭头函数没有自己的 this，箭头函数的 this 是继承而来，箭头函数的 this 由外层作用域决定。箭头函数没有自己的 this，它不能用作构造函数，不能使用 new 命令，否则会抛出一个错误。箭头函数没有自己的 arguments 对象，它使用的是外层函数的 arguments 对象。箭头函数没有原型属性。

```js
function foo() {
  return () => {
    return () => {
      console.log("id:", this.id);
    };
  };
}
var f = foo.call({ id: 1 });
var t = f.call({ id: 2 }); // id: 1
```

### 严格模式

在严格模式下，this 为 undefined，而不是全局对象。

```js
function foo() {
  "use strict";
  console.log(this);
}
foo(); // undefined
```
